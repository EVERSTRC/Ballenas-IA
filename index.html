if(rates[i].close < stLine[i]) { trend[i] = -1; stLine[i] = upBand; } else trend[i] = 1; } else { stLine[i] = MathMin(upBand, stLine[i + 1]); if(rates[i].close > stLine[i]) { trend[i] = 1; stLine[i] = dnBand; } else trend[i] = -1; } // 1. Señales Supertrend (ST BUY / ST SELL) if(trend[i] != trend[i + 1]) { string stName = prefix + "ST_ARR_" + TimeToString(rates[i].time); if(trend[i] == 1) CreateSignalText(stName, rates[i].time, rates[i].low - (atr * 0.4), "ST BUY", C'0,230,118', 233); else CreateSignalText(stName, rates[i].time, rates[i].high + (atr * 0.4), "ST SELL", C'255,23,68', 234); } // 2. Señales VSA (VSA BUY / VSA SELL en picos de volumen) double curVol = (double)rates[i].tick_volume; if(curVol > (avgVol * InpVSAFactor)) { double range = rates[i].high - rates[i].low; if(range > 0) { double buyRatio = (rates[i].close - rates[i].low) / range; string vsaName = prefix + "VSA_ARR_" + TimeToString(rates[i].time); if(buyRatio > 0.6) CreateSignalText(vsaName, rates[i].time, rates[i].low - (atr * 0.8), "VSA BUY", C'0,230,118', 233); else if(buyRatio < 0.4) CreateSignalText(vsaName, rates[i].time, rates[i].high + (atr * 0.8), "VSA SELL", C'255,23,68', 234); } } } // Actualizar Panel ObjectSetString(0, prefix + "ST_Signal", OBJPROP_TEXT, (trend[0] == 1) ? "BUY" : "SELL"); ObjectSetInteger(0, prefix + "ST_Signal", OBJPROP_COLOR, (trend[0] == 1) ? C'0,230,118' : C'255,23,68'); ObjectSetString(0, prefix + "ST_Price", OBJPROP_TEXT, "$" + DoubleToString(stLine[0], 2)); // Dibujar Niveles Entrada/SL/TP con etiquetas flotantes a la derecha DrawTradeSetupLevels(rates[0].close); } ObjectMove(0, lineName, 0, 0, price); ObjectSetInteger(0, lineName, OBJPROP_COLOR, col); } //+------------------------------------------------------------------+ //| VPVR DERECHA (PAGINADO A LA DERECHA DEL GRÁFICO) | //+------------------------------------------------------------------+ void DrawRightVPVR() { ObjectsDeleteAll(0, prefix + "VPVR_"); string bName = prefix + "VPVR_BAR_" + IntegerToString(j); ObjectCreate(0, bName, OBJ_RECTANGLE, 0, barStart, pHigh, barEnd, pLow); ObjectSetInteger(0, bName, OBJPROP_COLOR, (j == pocIndex) ?
