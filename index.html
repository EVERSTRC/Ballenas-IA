   ChartRedraw(0);   } //+------------------------------------------------------------------+ //| Custom indicator iteration function                              | //+------------------------------------------------------------------+ int OnCalculate(const int rates_total,                 const int prev_calculated,                 const datetime &time[],                 const double &open[],                 const double &high[],                 const double &low[],                 const double &close[],                 const long &tick_volume[],                 const long &volume[],                 const int &spread[])   {    UpdateDashboardData();    CalculateSupertrendSignals();        if(InpShowVPVR) DrawVPVR();    if(InpShowSetup) DrawTradeSetup();        return(rates_total);   } //+------------------------------------------------------------------+ //| MÓDULO SUPERTREND & FLECHAS EN GRÁFICO                           | //+------------------------------------------------------------------+ void CalculateSupertrendSignals()   {    int lookback = 200;    MqlRates rates[];    ArraySetAsSeries(rates, true);    if(CopyRates(_Symbol, _Period, 0, lookback, rates) < lookback) return;    double stLine[], trend[];    ArrayResize(stLine, lookback);    ArrayResize(trend, lookback);    ArrayInitialize(stLine, 0.0);    ArrayInitialize(trend, 1.0); // 1 = Bullish, -1 = Bearish    // Algoritmo Supertrend    for(int i = lookback - InpSTPeriod - 1; i >= 0; i--)      {       double atr = 0;       for(int k = 0; k < InpSTPeriod; k++) atr += (rates[i + k].high - rates[i + k].low);       atr /= InpSTPeriod;       double median = (rates[i].high + rates[i].low) / 2.0;       double upBand   = median + (InpSTMultiplier * atr);       double dnBand   = median - (InpSTMultiplier * atr);       double prevSt   = stLine[i + 1];       double prevTrend= trend[i + 1];       if(prevTrend == 1)         {          stLine[i] = MathMax(dnBand, prevSt);          if(rates[i].close < stLine[i])            {             trend[i] = -1;             stLine[i] = upBand;            }          else trend[i] = 1;         }       else         {          stLine[i] = MathMin(upBand, prevSt);          if(rates[i].close > stLine[i])            {             trend[i] = 1;             stLine[i] = dnBand;            }          else trend[i] = -1;         }       // Detectar Cambio de Tendencia (Señales ST BUY / ST SELL)       if(trend[i] != trend[i + 1])         {          string arrowName = prefix + "ST_ARR_" + TimeToString(rates[i].time);                    if(trend[i] == 1) // Nueva Señal Alcista            {             CreateArrow(arrowName, rates[i].time, rates[i].low - (atr * 0.5), 233, InpSTBuyColor, "ST BUY");            }          else // Nueva Señal Bajista            {             CreateArrow(arrowName, rates[i].time, rates[i].high + (atr * 0.5), 234, InpSTSellColor, "ST SELL");            }         }      }    // Actualizar Estado Supertrend en el Panel Izquierdo    string currSignal = (trend[0] == 1) ? "BUY" : "SELL";    color  currColor  = (trend[0] == 1) ? InpSTBuyColor : InpSTSellColor;    ObjectSetString(0, prefix + "ST_Signal", OBJPROP_TEXT, currSignal);    ObjectSetInteger(0, prefix + "ST_Signal", OBJPROP_COLOR, currColor);    ObjectSetString(0, prefix + "ST_Price", OBJPROP_TEXT, DoubleToString(stLine[0], _Digits));   } void CreateArrow(string name, datetime time, double price, int arrowCode, color arrowCol, string text)   {    if(ObjectFind(0, name) < 0)      {       ObjectCreate(0, name, OBJ_ARROW, 0, time, price);       ObjectSetInteger(0, name, OBJPROP_ARROWCODE, arrowCode);       ObjectSetInteger(0, name, OBJPROP_COLOR, arrowCol);       ObjectSetInteger(0, name, OBJPROP_WIDTH, 2);       ObjectSetInteger(0, name, OBJPROP_SELECTABLE, false);       ObjectSetString(0, name, OBJPROP_TEXT, text);      }   } //+------------------------------------------------------------------+ //| MÓDULO VPVR (Volume Profile Visible Range)                       | //+------------------------------------------------------------------+ void DrawVPVR()   {    ObjectsDeleteAll(0, prefix + "VPVR_");    MqlRates rates[];    ArraySetAsSeries(rates, true);    int copied = CopyRates(_Symbol, _Period, 0, InpVPVRBars, rates);    if(copied < InpVPVRBars) return;    double maxPrice = -1.0, minPrice = 999999.0;    for(int i = 0; i < copied; i++)      {       if(rates[i].high > maxPrice) maxPrice = rates[i].high;       if(rates[i].low < minPrice)  minPrice = rates[i].low;      }    if(maxPrice <= minPrice) return;    double step = (maxPrice - minPrice) / InpVPVRLevels;    double volBucket[];    ArrayResize(volBucket, InpVPVRLevels);    ArrayInitialize(volBucket, 0.0);    double maxVol = 0.0;    int pocIndex = 0;    for(int i = 0; i < copied; i++)      {       double mid = (rates[i].high + rates[i].low) / 2.0;       int idx = (int)((mid - minPrice) / step);       if(idx >= InpVPVRLevels) idx = InpVPVRLevels - 1;       if(idx < 0) idx = 0;       volBucket[idx] += (double)rates[i].tick_volume;       if(volBucket[idx] > maxVol)         {          maxVol = volBucket[idx];          pocIndex = idx;         }      }    datetime timeStart = rates[0].time;    for(int j = 0; j < InpVPVRLevels; j++)      {       if(volBucket[j] <= 0) continue;       double pLow  = minPrice + (j * step);       double pHigh = pLow + step;       int barLen = (int)((volBucket[j] / maxVol) * 20.0);        datetime barEnd = timeStart + PeriodSeconds(_Period) * MathMax(barLen, 1);       string bName = prefix + "VPVR_BAR_" + IntegerToString(j);       ObjectCreate(0, bName, OBJ_RECTANGLE, 0, timeStart, pHigh, barEnd, pLow);       ObjectSetInteger(0, bName, OBJPROP_COLOR, InpVPVRColor);       ObjectSetInteger(0, bName, OBJPROP_FILL, true);       ObjectSetInteger(0, bName, OBJPROP_BACK, true);       ObjectSetInteger(0, bName, OBJPROP_SELECTABLE, false);      }    double pocPrice = minPrice + (pocIndex * step) + (step / 2.0);    string pocName = prefix + "VPVR_POC";    ObjectCreate(0, pocName, OBJ_HLINE, 0, 0, pocPrice);    ObjectSetInteger(0, pocName, OBJPROP_COLOR, InpPOCColor);    ObjectSetInteger(0, pocName, OBJPROP_STYLE, STYLE_DASH);    ObjectSetInteger(0, pocName, OBJPROP_WIDTH, 1);    ObjectSetInteger(0, pocName, OBJPROP_SELECTABLE, false);    ObjectSetString(0, prefix + "SETUP_POC", OBJPROP_TEXT, DoubleToString(pocPrice, _Digits));   } //+------------------------------------------------------------------+ //| MÓDULO TRADE SETUP (ENTRY, SL, TP1, TP2)                         | //+------------------------------------------------------------------+ void DrawTradeSetup()   {    double closeCurr = iClose(_Symbol, _Period, 0);       atr = sumRange / 14.0;      }    if(atr <= 0) atr = _Point * 100;    double entry = closeCurr;    double sl    = closeCurr - (atr * 1.5);    double tp1   = closeCurr + ((entry - sl) * InpRiskReward);    double tp2   = closeCurr + ((entry - sl) * (InpRiskReward * 2.0));    CreateHorizontalLevel(prefix + "LVL_ENTRY", entry, "ENTRY", C'255,215,0');    CreateHorizontalLevel(prefix + "LVL_SL",    sl,    "SL",    C'255,23,68');    ObjectSetString(0, prefix + "SETUP_SL",    OBJPROP_TEXT, DoubleToString(sl, _Digits));    ObjectSetString(0, prefix + "SETUP_TP1",   OBJPROP_TEXT, DoubleToString(tp1, _Digits));    ObjectSetString(0, prefix + "SETUP_TP2",   OBJPROP_TEXT, DoubleToString(tp2, _Digits));   } void CreateHorizontalLevel(string name, double price, string labelText, color lineCol)   {    if(ObjectFind(0, name) < 0)      {       ObjectCreate(0, name, OBJ_HLINE, 0, 0, price);       ObjectSetInteger(0, name, OBJPROP_STYLE, STYLE_DASHDOT);
